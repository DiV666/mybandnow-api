## `src/Contexts/Configs/Communicator/domain/Communicator.ts`

```typescript
import { AggregateRoot } from '@Contexts/Shared/domain/AggregateRoot.js';
import { CommunicatorId } from './value-object/CommunicatorId.js';
import { CommunicatorCreatedAt } from './value-object/CommunicatorCreatedAt.js';
import { CommunicatorName } from './value-object/CommunicatorName.js';
import { CommunicatorApiUrl } from './value-object/CommunicatorApiUrl.js';

export interface CommunicatorPrimitives {
  id: string;
  createdAt: string;
  name: string;
  apiUrl: string | null;
}

export class Communicator extends AggregateRoot {
  constructor(
    readonly id: CommunicatorId,
    readonly createdAt: CommunicatorCreatedAt,
    readonly name: CommunicatorName,
    readonly apiUrl: CommunicatorApiUrl | null
  ) {
    super();
  }

  static fromPrimitives(plainData: CommunicatorPrimitives): Communicator {
    return new Communicator(
      new CommunicatorId(plainData.id),
      new CommunicatorCreatedAt(plainData.createdAt),
      new CommunicatorName(plainData.name),
      plainData.apiUrl ? new CommunicatorApiUrl(plainData.apiUrl) : null
    );
  }

  toPrimitives(): CommunicatorPrimitives {
    return {
      id: this.id.value,
      createdAt: this.createdAt.value.toISOString(),
      name: this.name.value,
      apiUrl: this.apiUrl?.value ?? null
    };
  }
}
```

> `static create()`, `update()`, and `remove()` are added automatically when the
> corresponding use cases are generated (`rb g uc`). They are NOT part of `rb g a`.
